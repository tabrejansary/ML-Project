## 2. Parse Filename & Build metadata.csv

Example filename: `SOB_B_A_14-22549AB-40-001.png`
- `SOB` = Slide On Board
- `B` = Benign (or `M` = Malignant)  
- `A` = Adenosis (subtype code)
- `14-22549AB` = Patient ID
- `40` = Magnification factor
- `001` = Image index

--- Cell 6 [CODE] ---
records = []

for class_name in ['benign', 'malignant']:
    class_dir = DATASET_DIR / class_name / 'SOB'
    if not class_dir.exists():
        print(f'⚠️  Not found: {class_dir}')
        continue

    for subtype_dir in sorted(class_dir.iterdir()):
        if not subtype_dir.is_dir():
            continue
        subtype = subtype_dir.name   # e.g. 'adenosis'

        for patient_dir in sorted(subtype_dir.iterdir()):
            if not patient_dir.is_dir():
                continue
            patient_id = patient_dir.name   # e.g. 'SOB_B_A_14-22549AB'

            for mag_dir in sorted(patient_dir.iterdir()):
                if not mag_dir.is_dir():
                    continue
                magnification = mag_dir.name   # e.g. '40X'

                for img_path in sorted(mag_dir.glob('*.png')):
                    records.append({
                        'image_path':    str(img_path),
                        'patient_id':    patient_id,
                        'class_name':    class_name,
                        'label':         CLASS_MAP[class_name],
                        'subtype':       subtype,
                        'magnification': magnification,
                    })

metadata = pd.DataFrame(records)

# ✂️ Notebook 02 — Patient-Level Data Splitting

**Phase 3 of the project**

**The most important rule:** Split PATIENTS, not images.

Why? One patient has ~96 images. If Patient P001's images appear in both
Train and Test, the model 'already knows' that patient → fake high accuracy.
This is called **Data Leakage**.

What this notebook does:
1. Loads metadata.csv
2. Gets the list of all unique patients
3. Splits patients into Train / Validation / Test (70 / 15 / 15)
4. Assigns every image to its patient's split
5. Saves the split info back to metadata.csv
6. Verifies no patient appears in more than one split

**Run each cell with Shift+Enter**

--- Cell 7 [MARKDOWN] ---
## 3. Stratified Patient Split

Split: **70% Train | 15% Validation | 15% Test**

We use `stratify=class_name` to ensure both splits have
roughly the same ratio of benign:malignant patients.

--- Cell 8 [CODE] ---
# Step 1: Split off 70% train, 30% temp
train_patients, temp_patients = train_test_split(
    patient_df,
    test_size   = 0.30,
    stratify    = patient_df['class_name'],
    random_state= RANDOM_SEED
)

# Step 2: Split the 30% temp into 15% val, 15% test
val_patients, test_patients = train_test_split(
    temp_patients,
    test_size   = 0.50,
    stratify    = temp_patients['class_name'],
    random_state= RANDOM_SEED
)

print('Patient Split Results:')

train_ids = set(train_patients['patient_id'])
val_ids   = set(val_patients['patient_id'])

train_val_overlap  = train_ids & val_ids
train_test_overlap = train_ids & test_ids
val_test_overlap   = val_ids   & test_ids

print('Leakage Check:')
print(f'  Train & Val  overlap : {len(train_val_overlap)}  (must be 0)')

assert len(val_test_overlap) == 0,   'LEAKAGE DETECTED between Val and Test!'

print('No leakage detected! Patient split is clean.')

--- Cell 2 [MARKDOWN] ---
## 1. Load Data — Prepare 3 Feature Sets

--- Cell 3 [CODE] ---
df = pd.read_csv(FEATURES_FUSED_CSV)

glcm_cols  = [c for c in df.columns if c.startswith('glcm_')]
lbp_cols   = [c for c in df.columns if c.startswith('lbp_')]
fused_cols = glcm_cols + lbp_cols

# Train+Val for CV, Test for final
tv_df   = df[df['split'].isin(['train','val'])]
test_df = df[df['split'] == 'test']

feature_sets = {
    'GLCM Only'    : glcm_cols,
    'LBP Only'     : lbp_cols,
    'GLCM + LBP'   : fused_cols,
}

print('Feature sets:')
for name, cols in feature_sets.items():
    print(f'  {name:15s}: {len(cols)} features')
print(f'\nTrain+Val: {len(tv_df)} images')
print(f'Test     : {len(test_df)} images')

--- Cell 4 [MARKDOWN] ---
## 2. Ablation with SVM (Best Classifier)

--- Cell 5 [CODE] ---
cv = StratifiedGroupKFold(n_splits=5)
g_tv = tv_df['patient_id'].values
y_tv = tv_df['label'].values
y_te = test_df['label'].values

ablation_results = []

for feat_name, feat_cols in feature_sets.items():
    X_tv = tv_df[feat_cols].values.astype(np.float32)
    X_te = test_df[feat_cols].values.astype(np.float32)

    k = min(30, len(feat_cols))

    svm_pipe = Pipeline([
        ('scaler', StandardScaler()),
        ('select', SelectKBest(f_classif, k=k)),
        ('clf', SVC(kernel='rbf', class_weight='balanced',
                    C=1, gamma='scale', probability=True,
                    random_state=RANDOM_SEED))
    ])

    rf_pipe = Pipeline([
        ('scaler', StandardScaler()),
        ('select', SelectKBest(f_classif, k=k)),
        ('clf', RandomForestClassifier(
            n_estimators=200, class_weight='balanced',
            max_depth=15, random_state=RANDOM_SEED, n_jobs=-1))
    ])

    for clf_name, pipe in [('SVM', svm_pipe), ('Random Forest', rf_pipe)]:
        # CV F1
        cv_f1 = cross_val_score(pipe, X_tv, y_tv,
                                groups=g_tv, cv=cv,
                                scoring='f1', n_jobs=-1)

        # Test
        pipe.fit(X_tv, y_tv)
        y_pred = pipe.predict(X_te)
        y_prob = pipe.predict_proba(X_te)[:,1]

        result = {
            'Features'   : feat_name,
            'Classifier' : clf_name,
            'N_Features' : len(feat_cols),
            'CV_F1'      : round(cv_f1.mean()*100, 2),
            'CV_F1_std'  : round(cv_f1.std()*100, 2),
            'Test_Acc'   : round(accuracy_score(y_te, y_pred)*100, 2),
            'Test_F1'    : round(f1_score(y_te, y_pred)*100, 2),
            'Test_AUC'   : round(roc_auc_score(y_te, y_prob)*100, 2),
        }
        ablation_results.append(result)
        print(f'  {feat_name:15s} | {clf_name:15s}: CV F1={result["CV_F1"]}%  Test F1={result["Test_F1"]}%  AUC={result["Test_AUC"]}%')

ablation_df = pd.DataFrame(ablation_results)
ablation_df.to_csv(RESULTS_DIR / 'ablation_results.csv', index=False)
print('\nAblation results saved!')
